```python
# 双向跨模态注意力机制
def forward(self, visual_proj, text_proj):
    # Vis → Text 注意力
    vis2text_attn = self.vis2text_attention(visual_proj, text_proj, text_proj)

    # Text → Vis 注意力
    text2vis_attn = self.text2vis_attention(text_proj, visual_proj, visual_proj)

    # 融合策略 (attention/concat/residual)
    if self.fusion_method == "attention":
        enhanced = visual_proj + vis2text_attn[0]
    elif self.fusion_method == "residual":
        enhanced = visual_proj + self.weight * (vis2text_attn[0] - visual_proj)

    return self.ffn(enhanced) + enhanced  # 前馈网络 + 残差
```
