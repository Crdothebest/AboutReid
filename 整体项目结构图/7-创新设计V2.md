│   └── 输出: 文本增强的视觉特征

│   ├── torch.cat([visual, text], dim=-1)
│   ├── 降维: Linear(1024→512)
│   └── 输出: 融合特征
│
└── 方法3: 残差增强
    ├── visual + text_guidance
    ├── 保持原始视觉信息
    └── 文本作为补充信号

```python
class TextVisionFusion(nn.Module):
    def __init__(self):
        super().__init__()
        # 加载预编码文本特征
        self.text_features = self.load_qwen_text_features()

        # 注意力融合
        self.vis2text_attn = nn.MultiheadAttention(512, 8)
        self.fusion_mlp = nn.Sequential(
            nn.Linear(1024, 512),
            nn.LayerNorm(512),
            nn.GELU()
        )

    def forward(self, visual_tokens, image_paths):
        # 获取对应的文本特征
        batch_text_features = []
        for path in image_paths:
            text_feat = self.text_features[path]  # [512]
            batch_text_features.append(text_feat)
        text_features = torch.stack(batch_text_features)  # [B, 512]

        # 注意力融合
        vis_enhanced, _ = self.vis2text_attn(
            visual_tokens, text_features, text_features
        )

        # 特征拼接融合
        combined = torch.cat([visual_tokens, vis_enhanced], dim=-1)
        fused = self.fusion_mlp(combined)

        return fused
```

class QwenVLTextLoader:
    def __init__(self, anno_dir="QwenVL_Anno"):
        self.text_features = {}
        for modality in ['RGB', 'NIR', 'TIR']:
            json_path = f"{anno_dir}/train_{modality}.json"
            with open(json_path, 'r') as f:
                data = json.load(f)
                for img_path, info in data.items():
                    key = f"{modality}_{img_path}"
                    self.text_features[key] = torch.tensor(info['text_emb'])

    def get_text_feature(self, img_path, modality):
        key = f"{modality}_{img_path}"
        return self.text_features.get(key, torch.zeros(512))

class EnhancedRGBNTDataset(Dataset):
    def __init__(self, text_loader=None, use_text_features=False):
        super().__init__()
        # 🔑 开关控制: 是否使用文本特征
        self.use_text_features = use_text_features
        self.text_loader = text_loader if use_text_features else None

    def __getitem__(self, idx):
        # 原有视觉数据加载 (始终执行)
        img_path, rgb_img, nir_img, tir_img, pid, camid, viewid = self._get_visual_data(idx)

        data_dict = {
            'img_path': img_path,
            'rgb_img': rgb_img, 'nir_img': nir_img, 'tir_img': tir_img,
            'pid': pid, 'camid': camid, 'viewid': viewid
        }

        # 条件分支: 文本特征加载
        if self.use_text_features and self.text_loader is not None:
            # ✅ 启用文本特征: 加载对应的文本特征
            rgb_text = self.text_loader.get_text_feature(img_path, 'RGB')  # [512]
            nir_text = self.text_loader.get_text_feature(img_path, 'NIR')  # [512]
            tir_text = self.text_loader.get_text_feature(img_path, 'TIR')  # [512]

            data_dict.update({
                'rgb_text': rgb_text,
                'nir_text': nir_text,
                'tir_text': tir_text
            })
        else:
            # ❌ 不使用文本特征: 返回零张量占位符 (保持接口一致)
            dummy_text = torch.zeros(512)
            data_dict.update({
                'rgb_text': dummy_text,
                'nir_text': dummy_text,
                'tir_text': dummy_text
            })

        return data_dict

class EnhancedMambaPro(nn.Module):
    def __init__(self, use_text_fusion=False, text_fusion_method="attention"):
        super().__init__()
        # 🔑 开关参数
        self.use_text_fusion = use_text_fusion
        self.text_fusion_method = text_fusion_method

        # ============ 始终存在的核心模块 ============
        # 原有视觉处理 (RGB/NIR/TIR分支)
        self.rgb_branch = CLIPVisionTransformer(modality='rgb')
        self.nir_branch = CLIPVisionTransformer(modality='nir')
        self.tir_branch = CLIPVisionTransformer(modality='tir')

        # 多尺度MoE处理 (视觉增强)
        self.multi_scale_moe = MultiScaleMoE()

        # Mamba聚合 (多模态融合)
        self.mamba_aggregation = MambaAggregation()

        # ============ 条件模块: 仅在启用文本融合时创建 ============
        if self.use_text_fusion:
            if self.text_fusion_method == "attention":
                self.text_fusion = CrossModalAttentionFusion()
            elif self.text_fusion_method == "concat":
                self.text_fusion = TextConcatFusion()
            elif self.text_fusion_method == "residual":
                self.text_fusion = TextResidualFusion()
            else:
                raise ValueError(f"Unknown text fusion method: {self.text_fusion_method}")
        else:
            # 文本融合关闭时，相关模块为None
            self.text_fusion = None

    def forward(self, batch):
        # ============ Phase 1: 视觉特征提取 (始终执行) ============
        rgb_tokens = self.rgb_branch(batch['rgb_img'])  # [B,129,512]
        nir_tokens = self.nir_branch(batch['nir_img'])  # [B,129,512]
        tir_tokens = self.tir_branch(batch['tir_img'])  # [B,129,512]

        # ============ Phase 2: 多尺度增强 (始终执行) ============
        rgb_enhanced = self.multi_scale_moe(rgb_tokens)  # [B,129,512]
        nir_enhanced = self.multi_scale_moe(nir_tokens)  # [B,129,512]
        tir_enhanced = self.multi_scale_moe(tir_tokens)  # [B,129,512]

        # ============ Phase 3: 条件分支 - 文本融合开关 ============
        if self.use_text_fusion and self.text_fusion is not None:
            # ✅ 开关开启: 执行跨模态融合
            rgb_fused = self.text_fusion(rgb_enhanced, batch['rgb_text'])  # [B,512]
            nir_fused = self.text_fusion(nir_enhanced, batch['nir_text'])  # [B,512]
            tir_fused = self.text_fusion(tir_enhanced, batch['tir_text'])  # [B,512]
        else:
            # ❌ 开关关闭: 保持原AboutReid结构
            # 直接使用增强后的视觉特征 (CLS token)
            rgb_fused = rgb_enhanced[:, 0]  # [B,512] - CLS token
            nir_fused = nir_enhanced[:, 0]  # [B,512] - CLS token
            tir_fused = tir_enhanced[:, 0]  # [B,512] - CLS token

        # ============ Phase 4: Mamba聚合融合 (始终执行) ============
        final_features = self.mamba_aggregation([rgb_fused, nir_fused, tir_fused])

        return final_features

class CrossModalAttentionFusion(nn.Module):
    def __init__(self, embed_dim=512, num_heads=8):
        super().__init__()
        # 视觉到文本的注意力
        self.vis2text_attention = nn.MultiheadAttention(embed_dim, num_heads)
        # 文本到视觉的注意力
        self.text2vis_attention = nn.MultiheadAttention(embed_dim, num_heads)
        # 特征融合MLP
        self.fusion_mlp = nn.Sequential(
            nn.Linear(embed_dim * 2, embed_dim),
            nn.LayerNorm(embed_dim),
            nn.GELU(),
            nn.Linear(embed_dim, embed_dim)
        )

    def forward(self, visual_tokens, text_feat):
        # 全局视觉特征 (CLS token)
        visual_global = visual_tokens[:, 0]  # [B, 512]

        # 视觉引导文本注意力
        vis_enhanced_text, _ = self.vis2text_attention(
            text_feat.unsqueeze(0), visual_global.unsqueeze(0), visual_global.unsqueeze(0)
        )
        # 文本引导视觉注意力
        text_enhanced_vis, _ = self.text2vis_attention(
            visual_global.unsqueeze(0), text_feat.unsqueeze(0), text_feat.unsqueeze(0)
        )

        # 特征融合
        combined = torch.cat([text_enhanced_vis.squeeze(0), vis_enhanced_text.squeeze(0)], dim=-1)
        fused = self.fusion_mlp(combined)

        return fused

class TextConcatFusion(nn.Module):
    """特征拼接融合方法 - 简单高效"""
    def __init__(self, embed_dim=512):
        super().__init__()
        self.fusion_mlp = nn.Sequential(
            nn.Linear(embed_dim * 2, embed_dim),
            nn.LayerNorm(embed_dim),
            nn.GELU(),
            nn.Linear(embed_dim, embed_dim)
        )

    def forward(self, visual_tokens, text_feat):
        # 提取视觉全局特征
        visual_global = visual_tokens[:, 0]  # [B, 512]

        # 直接拼接视觉和文本特征
        combined = torch.cat([visual_global, text_feat], dim=-1)  # [B, 1024]

        # MLP融合降维
        return self.fusion_mlp(combined)  # [B, 512]

class TextResidualFusion(nn.Module):
    """残差增强融合方法 - 保留原始视觉信息"""
    def __init__(self, embed_dim=512, fusion_weight=0.3):
        super().__init__()
        self.fusion_weight = fusion_weight
        self.adapter = nn.Sequential(
            nn.Linear(embed_dim, embed_dim // 2),
            nn.GELU(),
            nn.Linear(embed_dim // 2, embed_dim),
            nn.LayerNorm(embed_dim)
        )

    def forward(self, visual_tokens, text_feat):
        # 提取视觉全局特征
        visual_global = visual_tokens[:, 0]  # [B, 512]

        # 文本引导的残差增强
        text_guidance = self.adapter(text_feat)  # [B, 512]

        # 残差融合: 视觉特征 + 文本引导 (可控权重)
        enhanced = visual_global + self.fusion_weight * text_guidance

        return enhanced  # [B, 512]
